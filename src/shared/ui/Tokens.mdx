# Tokens

Defined once in `src/shared/ui/tokens.css`; every rule and component uses them.

## Colors

export const Swatch = ({ name, use }) => (
  <div style={{ display: "flex", alignItems: "center", gap: 12, margin: "6px 0" }}>

    <code>{name}</code>
    <span style={{ color: "var(--color-muted)" }}>{use}</span>
  </div>
